package main

import (
 "net/http"
 "os"
 "path/filepath"
 "testing"
 "time"
)

func TestBrowserDemoHarness(t *testing.T) {
 if os.Getenv("SLOTH_BROWSER_HARNESS")!="1" { t.Skip("isolated browser fixture only") }
 base:=os.Getenv("SLOTH_DEMO_BASE")
 if base=="" {t.Fatal("missing isolated data root")}
 if err:=setPreferredQuant(base,"Q4_K_M");err!=nil{t.Fatal(err)}
 store,err:=openStore(base);if err!=nil{t.Fatal(err)}
 // This fixed setup key protects fictional test data only and is never used
 // by a production connector or a real user's browser pairing.
 app:=newBridge(base,store,"sloth-synthetic-demo-key-for-local-testing-2026")
 server:=&http.Server{Addr:"127.0.0.1:39872",Handler:app.handler(),ReadHeaderTimeout:5*time.Second}
 go func(){if err:=server.ListenAndServe();err!=http.ErrServerClosed{t.Error(err)}}()
 defer server.Close()
 t.Log("Isolated browser connector ready; no user notes or accounts loaded")
 deadline:=time.Now().Add(12*time.Minute)
 for time.Now().Before(deadline){
  if _,err:=os.Stat(filepath.Join(base,"stop-harness"));err==nil{return}
  time.Sleep(200*time.Millisecond)
 }
}
